-- =============================================
-- 1. Создание базы данных
-- =============================================
CREATE DATABASE IF NOT EXISTS orderdb 
CHARACTER SET utf8mb4 
COLLATE utf8mb4_unicode_ci;

USE orderdb;

-- =============================================
-- 2. Создание таблиц
-- =============================================

-- 2.1. Таблица пользователей
CREATE TABLE IF NOT EXISTS Users (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    Login VARCHAR(100) NOT NULL UNIQUE,
    Password VARCHAR(255) NOT NULL,
    Email VARCHAR(255) UNIQUE,
    Firstname VARCHAR(100),
    Lastname VARCHAR(100),
    Photo LONGBLOB,
    Token VARCHAR(500),
    Role VARCHAR(50) NOT NULL DEFAULT 'User',
    INDEX idx_token (Token),
    INDEX idx_login (Login),
    INDEX idx_email (Email)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 2.2. Таблица городов
CREATE TABLE IF NOT EXISTS Cities (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    Name VARCHAR(100) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 2.3. Таблица отделов
CREATE TABLE IF NOT EXISTS Departments (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    Name VARCHAR(255) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 2.4. Таблица должностей
CREATE TABLE IF NOT EXISTS Positions (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    Name VARCHAR(255) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 2.5. Таблица статусов кандидатов
CREATE TABLE IF NOT EXISTS CandidateStatuses (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    Name VARCHAR(100) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 2.6. Таблица статусов удаления
CREATE TABLE IF NOT EXISTS RemoveStatuses (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    Name VARCHAR(100) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 2.7. Таблица сотрудников
CREATE TABLE IF NOT EXISTS Employees (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    Name VARCHAR(255),
    Email VARCHAR(255),
    Phone VARCHAR(50),
    IdDepartment INT,
    FOREIGN KEY (IdDepartment) REFERENCES Departments(Id) ON DELETE SET NULL,
    INDEX idx_department (IdDepartment)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 2.8. Таблица вакансий
CREATE TABLE IF NOT EXISTS Vacancies (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    Title VARCHAR(500) NOT NULL,
    IdDepartment INT,
    OpeningDate DATETIME,
    IdPosition INT,
    SalaryRange VARCHAR(100),
    RequiredQuantity INT DEFAULT 0,
    Experience VARCHAR(100),
    IdCity INT,
    IsPublic BOOLEAN DEFAULT FALSE,
    Description TEXT,
    TeamOfEmployment VARCHAR(1000),
    Requirements TEXT,
    TechnicalQuestions TEXT,
    TestTask TEXT,
    IdSpecialist INT,
    FOREIGN KEY (IdDepartment) REFERENCES Departments(Id) ON DELETE SET NULL,
    FOREIGN KEY (IdPosition) REFERENCES Positions(Id) ON DELETE SET NULL,
    FOREIGN KEY (IdCity) REFERENCES Cities(Id) ON DELETE SET NULL,
    FOREIGN KEY (IdSpecialist) REFERENCES Employees(Id) ON DELETE SET NULL,
    INDEX idx_department (IdDepartment),
    INDEX idx_position (IdPosition),
    INDEX idx_city (IdCity),
    INDEX idx_specialist (IdSpecialist)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 2.9. Таблица кандидатов
CREATE TABLE IF NOT EXISTS Candidates (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    Name VARCHAR(255),
    Email VARCHAR(255),
    Phone VARCHAR(50),
    IdDepartment INT,
    IdStatus INT,
    IdCity INT,
    IdVacancy INT,
    IdHR INT,
    FOREIGN KEY (IdDepartment) REFERENCES Departments(Id) ON DELETE SET NULL,
    FOREIGN KEY (IdStatus) REFERENCES CandidateStatuses(Id) ON DELETE SET NULL,
    FOREIGN KEY (IdCity) REFERENCES Cities(Id) ON DELETE SET NULL,
    FOREIGN KEY (IdVacancy) REFERENCES Vacancies(Id) ON DELETE SET NULL,
    FOREIGN KEY (IdHR) REFERENCES Employees(Id) ON DELETE SET NULL,
    INDEX idx_status (IdStatus),
    INDEX idx_city (IdCity),
    INDEX idx_vacancy (IdVacancy),
    INDEX idx_hr (IdHR)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 2.10. Таблица комментариев к кандидатам
CREATE TABLE IF NOT EXISTS CandidateComments (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    IdCandidate INT NOT NULL,
    Comment TEXT,
    FOREIGN KEY (IdCandidate) REFERENCES Candidates(Id) ON DELETE CASCADE,
    INDEX idx_candidate (IdCandidate)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 2.11. Таблица файлов вакансий
CREATE TABLE IF NOT EXISTS VacancyFiles (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    IdVacancy INT NOT NULL,
    FileName VARCHAR(500),
    FileData LONGBLOB,
    FOREIGN KEY (IdVacancy) REFERENCES Vacancies(Id) ON DELETE CASCADE,
    INDEX idx_vacancy (IdVacancy)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
